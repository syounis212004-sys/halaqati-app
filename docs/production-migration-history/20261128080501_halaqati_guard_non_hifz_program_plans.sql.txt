CREATE SCHEMA IF NOT EXISTS halaqati_repair_private;
REVOKE ALL ON SCHEMA halaqati_repair_private FROM PUBLIC, anon, authenticated;
CREATE FUNCTION halaqati_repair_private.guard_plan_program_scope() RETURNS trigger
LANGUAGE plpgsql SECURITY DEFINER SET search_path=pg_catalog,public AS $guard$
BEGIN
 IF EXISTS (SELECT 1 FROM public.halaqas h WHERE h.id=NEW.halaqa_id AND coalesce(h.halaqa_type,'regular')<>'regular') THEN
   RAISE EXCEPTION 'خطط الحفظ غير متاحة لهذا البرنامج في الإصدار الحالي' USING ERRCODE='23514';
 END IF;
 RETURN NEW;
END $guard$;
REVOKE ALL ON FUNCTION halaqati_repair_private.guard_plan_program_scope() FROM PUBLIC, anon, authenticated;
CREATE TRIGGER zz_halaqati_guard_plan_program_scope BEFORE INSERT OR UPDATE OF halaqa_id ON public.smart_plans FOR EACH ROW EXECUTE FUNCTION halaqati_repair_private.guard_plan_program_scope();
