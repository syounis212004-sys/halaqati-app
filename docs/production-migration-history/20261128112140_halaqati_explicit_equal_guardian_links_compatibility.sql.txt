CREATE OR REPLACE FUNCTION public.rc_review_request(p_kind text, p_id uuid, p_action text, p_reason text DEFAULT NULL::text)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  uid uuid := auth.uid();
  r text := public.current_role()::text;
  tp public.profiles%rowtype;
  l public.guardian_link_requests%rowtype;
  v_reason text := nullif(trim(coalesce(p_reason, '')), '');
  student public.students%rowtype;
  scope_id uuid := public.my_association();
  owner boolean := public.is_platform_owner();
  a record;
begin
  if uid is null or r not in ('admin', 'supervisor', 'teacher') or not exists (
    select 1 from public.profiles reviewer
    where reviewer.id = uid and reviewer.active and not reviewer.frozen
      and reviewer.approval_status = 'approved'
  ) then
    raise exception 'لا تملك صلاحية مراجعة الطلبات';
  end if;
  if p_action is null or p_action not in ('approve', 'reject') then raise exception 'إجراء غير صالح'; end if;

  if p_kind = 'account' then
    select * into tp from public.profiles where id = p_id for update;
    if not found then return jsonb_build_object('ok', false, 'error', 'الطلب غير موجود'); end if;
    if tp.id = uid or (
      (r = 'admin' or (r = 'supervisor' and tp.requested_role::text = 'teacher'))
      and (owner or tp.association_id = scope_id)
    ) is not true then
      raise exception 'لا تملك صلاحية على هذا الطلب' using errcode = '42501';
    end if;
    if tp.approval_status <> 'pending' or tp.frozen or not tp.onboarded then
      return jsonb_build_object('ok', false, 'error', 'تمت معالجة هذا الطلب مسبقًا');
    end if;
    if coalesce(tp.requested_role, tp.role)::text = 'guardian' then
      return jsonb_build_object('ok', false, 'error', 'راجع طلب الربط الخاص بولي الأمر');
    end if;
    if tp.requested_role is null or tp.requested_role::text not in ('teacher', 'supervisor', 'observer') then
      raise exception 'نوع الحساب المطلوب غير صالح';
    end if;

    perform set_config('halaqati.review_ok', '1', true);
    if p_action = 'approve' then
      update public.profiles
         set role = tp.requested_role, approval_status = 'approved', active = true,
             approved_at = now(), approved_by = uid, rejection_reason = null
       where id = p_id;
      perform public.rc_approval_notify(p_id, 'تم اعتماد حسابك', 'تم اعتماد حسابك. يمكنك الآن استخدام التطبيق.',
        'acct-ok:' || p_id::text || ':' || to_char(now(), 'YYYYMMDDHH24MISS'));
    else
      update public.profiles
         set approval_status = 'rejected', active = false, rejection_reason = v_reason
       where id = p_id;
    end if;
    perform set_config('halaqati.review_ok', '', true);
    perform public.rc_audit('profiles', p_id::text, 'rc_review_account_' || p_action,
      jsonb_build_object('by_role', r, 'role', tp.requested_role::text, 'reason', v_reason));
    return jsonb_build_object('ok', true);

  elsif p_kind = 'link' then
    select * into l from public.guardian_link_requests where id = p_id for update;
    if not found then return jsonb_build_object('ok', false, 'error', 'الطلب غير موجود'); end if;
    select * into tp from public.profiles where id = l.guardian_id for update;
    select * into student from public.students where id = l.student_id for update;
    if student.id is null or tp.id is null or tp.id = uid
       or not (r = 'admin' or (r = 'teacher' and public.rc_is_student_teacher(uid, l.student_id)))
       or not (owner or coalesce(student.association_id = scope_id, false))
       or (tp.association_id is not null and tp.association_id is distinct from student.association_id) then
      raise exception 'لا تملك صلاحية على هذا الطلب' using errcode = '42501';
    end if;
    if p_action = 'approve' and (
      student.status <> 'active' or tp.frozen or not tp.onboarded
      or not ((tp.approval_status = 'pending' and tp.requested_role::text = 'guardian')
           or (tp.approval_status = 'approved' and tp.role::text = 'guardian' and tp.active))
    ) is not false then
      return jsonb_build_object('ok', false, 'error', 'حالة الطالب أو حساب ولي الأمر لا تسمح بالربط');
    end if;
    if r = 'admin' and p_action = 'approve' and exists (
      select 1 from public.student_guardians sg join public.students child on child.id = sg.student_id
      where sg.guardian_id = tp.id and child.association_id is distinct from student.association_id
    ) then
      return jsonb_build_object('ok', false, 'error', 'يرتبط هذا الحساب بطالب في جمعية أخرى');
    end if;

    if r = 'teacher' then
      if l.status <> 'pending' then
        return jsonb_build_object('ok', false, 'error', 'تمت معالجة هذا الطلب مسبقًا');
      end if;
      if p_action = 'approve' then
        update public.guardian_link_requests
           set status = 'teacher_approved', teacher_id = uid, teacher_at = now() where id = p_id;
        for a in select id from public.profiles where role = 'admin' and approval_status = 'approved' and active and not frozen
          and (is_platform_owner or association_id = student.association_id) loop
          perform public.rc_approval_notify(a.id, 'وافق المحفّظ على طلب ولي أمر',
            'وافق المحفّظ على ربط ولي أمر بالطالب ' || coalesce(student.full_name, '') || '. بانتظار تأكيدك من «طلبات الاعتماد».',
            'link-tok:' || p_id::text);
        end loop;
      else
        update public.guardian_link_requests
           set status = 'rejected', teacher_id = uid, teacher_at = now(), reason = v_reason where id = p_id;
        perform public.rc_approval_notify(l.guardian_id, 'تم رفض طلب الربط',
          coalesce(v_reason, 'رفض محفّظ الحلقة طلب ربط حسابك بالطالب.'),
          'link-rej:' || p_id::text || ':' || to_char(now(), 'YYYYMMDDHH24MISS'));
      end if;

    elsif r = 'admin' then
      if l.status not in ('pending', 'teacher_approved') then
        return jsonb_build_object('ok', false, 'error', 'تمت معالجة هذا الطلب مسبقًا');
      end if;
      if p_action = 'approve' then
        update public.guardian_link_requests
           set status = 'approved', admin_id = uid, admin_at = now() where id = p_id;
        if tp.approval_status = 'pending' then
          perform set_config('halaqati.review_ok', '1', true);
          perform set_config('halaqati.guardian_link_approval', '1', true);
          update public.profiles
             set role = 'guardian', approval_status = 'approved', active = true,
                 approved_at = now(), approved_by = uid, rejection_reason = null
           where id = l.guardian_id;
          perform set_config('halaqati.guardian_link_approval', '', true);
          perform set_config('halaqati.review_ok', '', true);
        end if;
        insert into public.student_guardians (student_id, guardian_id, relation, is_primary)
        values (l.student_id, l.guardian_id, 'ولي أمر',
          not exists (select 1 from public.student_guardians existing where existing.student_id = l.student_id and existing.is_primary))
        on conflict (student_id, guardian_id) do nothing;
        perform public.rc_approval_notify(l.guardian_id, 'تم اعتماد طلب الربط',
          'تم ربط حسابك بالطالب ' || coalesce(student.full_name, '') || '. يمكنك متابعة ابنك الآن.',
          'link-ok:' || p_id::text);
      else
        update public.guardian_link_requests
           set status = 'rejected', admin_id = uid, admin_at = now(), reason = v_reason where id = p_id;
        perform public.rc_approval_notify(l.guardian_id, 'تم رفض طلب الربط',
          coalesce(v_reason, 'رفض مسؤول النظام طلب ربط حسابك بالطالب.'),
          'link-rej:' || p_id::text || ':' || to_char(now(), 'YYYYMMDDHH24MISS'));
      end if;
    else
      raise exception 'لا تملك صلاحية على هذا الطلب';
    end if;

    perform public.rc_audit('guardian_link_requests', p_id::text, 'rc_review_link_' || p_action,
      jsonb_build_object('by_role', r, 'student_id', l.student_id, 'guardian_id', l.guardian_id, 'reason', v_reason));
    return jsonb_build_object('ok', true);
  end if;

  raise exception 'نوع الطلب غير صالح';
end $function$
;
CREATE OR REPLACE FUNCTION public.rc_create_link_request(p_guardian uuid, p_national_id text, p_birth_date date)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  v_nid text;
  v_sid uuid;
  v_fails integer;
  v_status text;
  v_name text;
  v_scope uuid;
  t record;
begin
  select association_id into v_scope from public.profiles where id = p_guardian and not coalesce(frozen,false);
  if p_guardian is distinct from auth.uid() or v_scope is null then
    return jsonb_build_object('ok', false, 'error', 'تعذر التحقق من نطاق الحساب');
  end if;
  v_nid := regexp_replace(translate(coalesce(p_national_id, ''), '٠١٢٣٤٥٦٧٨٩', '0123456789'), '\s', '', 'g');
  if v_nid !~ '^[0-9]{5,20}$' then
    return jsonb_build_object('ok', false, 'error', 'رقم هوية الطالب غير صالح');
  end if;
  if p_birth_date is null then
    return jsonb_build_object('ok', false, 'error', 'تاريخ ميلاد الطالب مطلوب');
  end if;
  select count(*) into v_fails from public.rc_link_attempts
  where user_id = p_guardian and not ok and created_at > now() - interval '1 hour';
  if v_fails >= 5 then
    return jsonb_build_object('ok', false, 'error', 'محاولات كثيرة غير صحيحة. حاول بعد ساعة أو تواصل مع الإدارة');
  end if;

  select s.id, s.full_name into v_sid, v_name
  from public.students s
  where regexp_replace(translate(coalesce(s.national_id, ''), '٠١٢٣٤٥٦٧٨٩', '0123456789'), '\s', '', 'g') = v_nid
    and s.date_of_birth = p_birth_date
    and s.status = 'active' and s.association_id = v_scope
  limit 1;

  if v_sid is null then
    insert into public.rc_link_attempts(user_id, ok) values (p_guardian, false);
    return jsonb_build_object('ok', false, 'error', 'البيانات غير مطابقة لأي طالب مسجّل');
  end if;
  insert into public.rc_link_attempts(user_id, ok) values (p_guardian, true);

  if exists (select 1 from public.student_guardians where student_id = v_sid and guardian_id = p_guardian) then
    return jsonb_build_object('ok', false, 'error', 'هذا الطالب مرتبط بحسابك مسبقًا');
  end if;

  v_status := case when public.rc_student_has_teacher(v_sid) then 'pending' else 'teacher_approved' end;
  insert into public.guardian_link_requests (guardian_id, student_id, status)
  values (p_guardian, v_sid, v_status)
  on conflict (guardian_id, student_id) do update
    set status = excluded.status, reason = null, teacher_id = null, teacher_at = null, admin_id = null, admin_at = null,
        created_at = now()
    where public.guardian_link_requests.status = 'rejected';

  -- tell the people who must act next
  if v_status = 'pending' then
    for t in
      select distinct u.id from (
        select s.primary_teacher_id as id from public.students s where s.id = v_sid and s.primary_teacher_id is not null
        union
        select ht.teacher_id from public.students s join public.halaqa_teachers ht on ht.halaqa_id = s.halaqa_id where s.id = v_sid
      ) u
    loop
      perform public.rc_approval_notify(t.id, 'طلب ربط ولي أمر',
        'طلب ولي أمر جديد لربط حسابه بالطالب ' || v_name || '. راجعه من «طلبات الاعتماد».',
        'link-req:' || p_guardian::text || ':' || v_sid::text || ':t:' || to_char(now(), 'YYYYMMDDHH24MISS'));
    end loop;
  else
    for t in select id from public.profiles where role = 'admin' and approval_status = 'approved' and active and not frozen and (is_platform_owner or association_id = v_scope) loop
      perform public.rc_approval_notify(t.id, 'طلب ربط ولي أمر',
        'طلب ولي أمر جديد لربط حسابه بالطالب ' || v_name || ' (لا يوجد محفّظ للحلقة). راجعه من «طلبات الاعتماد».',
        'link-req:' || p_guardian::text || ':' || v_sid::text || ':a:' || to_char(now(), 'YYYYMMDDHH24MISS'));
    end loop;
  end if;

  return jsonb_build_object('ok', true, 'stage', case when v_status = 'pending' then 'teacher' else 'admin' end);
end $function$
;
-- Legacy single-guardian editor can safely save an unchanged relationship, but cannot
-- create an unapproved link, replace another guardian, or bulk-delete links.
create or replace function public.set_student_primary_guardian(p_student_id uuid,p_guardian_id uuid default null)
returns void language plpgsql security definer set search_path='public' as $f$
declare s public.students%rowtype;
begin
  select * into s from public.students where id=p_student_id for update;
  if auth.uid() is null or s.id is null or not public.can_administer_student(p_student_id)
    or not (public.is_platform_owner() or (s.association_id is not null and s.association_id=public.my_association())) then
    raise exception 'لا تملك صلاحية إدارة ارتباطات هذا الطالب' using errcode='42501';
  end if;
  if p_guardian_id is null then
    if exists(select 1 from public.student_guardians where student_id=p_student_id) then
      raise exception 'إزالة الارتباطات تتطلب اختيار ولي الأمر المحدد من إدارة الارتباطات الجديدة' using errcode='22023';
    end if;
    return;
  end if;
  if exists(select 1 from public.student_guardians where student_id=p_student_id and guardian_id=p_guardian_id) then return; end if;
  raise exception 'يجب إرسال طلب ربط ولي الأمر واعتماده من طلبات الاعتماد' using errcode='42501';
end $f$;
-- Remove only an explicitly identified relationship; no deletion of other guardians.
create function public.rc_unlink_guardian(p_student_id uuid,p_guardian_id uuid,p_reason text)
returns jsonb language plpgsql security definer set search_path='public' as $f$
declare s public.students%rowtype; v_count integer; v_reason text:=nullif(trim(p_reason),'');
begin
  if auth.uid() is null or not public.is_admin() or not exists(select 1 from public.profiles p where p.id=auth.uid() and p.active and not p.frozen and p.approval_status='approved') then
    raise exception 'إزالة ارتباط ولي الأمر من صلاحية مسؤول النظام' using errcode='42501';
  end if;
  if v_reason is null then raise exception 'سبب إزالة الارتباط مطلوب' using errcode='22023'; end if;
  -- same lock order as review (request first, then student)
  perform 1 from public.guardian_link_requests where student_id=p_student_id and guardian_id=p_guardian_id for update;
  select * into s from public.students where id=p_student_id for update;
  if s.id is null or not (public.is_platform_owner() or (s.association_id is not null and s.association_id=public.my_association())) or not public.assoc_writable(s.association_id) then
    raise exception 'لا تملك صلاحية إدارة ارتباطات هذا الطالب' using errcode='42501';
  end if;
  delete from public.student_guardians where student_id=p_student_id and guardian_id=p_guardian_id;
  get diagnostics v_count=row_count;
  if v_count=0 then return jsonb_build_object('ok',true,'removed',false); end if;
  update public.guardian_link_requests set status='rejected',reason='إزالة الارتباط: '||v_reason,admin_id=auth.uid(),admin_at=now()
    where student_id=p_student_id and guardian_id=p_guardian_id and status in ('pending','teacher_approved','approved');
  -- maintain a single pointer for old clients; all remaining links retain equal access.
  if not exists(select 1 from public.student_guardians where student_id=p_student_id and is_primary) then
    update public.student_guardians set is_primary=true where student_id=p_student_id and guardian_id=(
      select guardian_id from public.student_guardians where student_id=p_student_id order by created_at,guardian_id limit 1);
  end if;
  perform public.rc_audit('student_guardians',p_student_id::text,'rc_unlink_guardian',jsonb_build_object('guardian_id',p_guardian_id,'reason',v_reason));
  return jsonb_build_object('ok',true,'removed',true);
end $f$;
revoke all on function public.rc_unlink_guardian(uuid,uuid,text) from public,anon,authenticated;
grant execute on function public.rc_unlink_guardian(uuid,uuid,text) to authenticated,service_role;
revoke insert,update,delete,truncate,trigger,references on public.student_guardians from public,anon,authenticated;
comment on column public.student_guardians.is_primary is 'Legacy single-guardian display pointer only. All explicitly approved guardian links have equal access; new clients must list all links. Remove this field after client migration.';
do $verify$
begin
  if exists(select 1 from halaqati_repair_private.guardian_explicit_links_backup_20261128 b where b.kind='link' and not exists(select 1 from public.student_guardians sg where to_jsonb(sg)=b.previous_value)) then raise exception 'Existing link changed unexpectedly'; end if;
  if has_table_privilege('authenticated','public.student_guardians','INSERT') or has_table_privilege('authenticated','public.student_guardians','UPDATE') or has_table_privilege('authenticated','public.student_guardians','DELETE') then raise exception 'Direct write privilege still present'; end if;
end $verify$;
